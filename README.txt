Folder structure:

gulsehar-site/
  index.html
  case.html
  vercel.json
  assets/
    site.css
    site.js
    cases.js
    (13 PNG files, names below)
  demos/      quickcast.html, rightsvault.html, snapsell.html, commentflow.html
  concepts/   hartwell-grey.html, velvet-hour.html, pinehaven.html
